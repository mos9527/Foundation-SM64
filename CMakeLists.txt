# ---------------------------------------------------------------------------
# Asset pipeline
#
# Every tool that generates assets is a *host* executable, so the whole pipeline
# can only run when host == target. For a cross build (Android) it is therefore
# configured and built in a nested host build tree that writes into the same
# SM64_GEN_DIR; this tree never builds the tools and only consumes whatever the
# host build produced.
# ---------------------------------------------------------------------------
file(MAKE_DIRECTORY "${SM64_GEN_DIR}" "${SM64_GEN_DIR}/include")

# Publishes the generated source lists so the cross build can pick them up
# without re-running the (glob-driven) asset rules.
function(sm64_write_generated_sources path)
  set(_body "# Generated by SM64_HOST_ASSET_BUILD -- do not edit.\n")
  foreach(_f IN LISTS SM64_GENERATED_C_FILES)
    string(APPEND _body "list(APPEND SM64_GENERATED_C_FILES \"${_f}\")\n")
  endforeach()
  foreach(_f IN LISTS SM64_ASSET_OUTPUTS)
    string(APPEND _body "list(APPEND SM64_ASSET_OUTPUTS \"${_f}\")\n")
  endforeach()
  file(WRITE "${path}" "${_body}")
endfunction()

function(sm64_run_host_asset_build)
  set(_host_args
    -S "${SM64_ROOT}"
    -B "${SM64_HOST_ASSET_BINARY_DIR}"
    -DSM64_HOST_ASSET_BUILD=ON
    -DSM64_GEN_DIR=${SM64_GEN_DIR}
    -DSM64_VERSION=${SM64_VERSION}
    -DSM64_GRUCODE=${SM64_GRUCODE}
    -DCMAKE_BUILD_TYPE=Release)
  if(SM64_HOST_C_COMPILER)
    list(APPEND _host_args -DCMAKE_C_COMPILER=${SM64_HOST_C_COMPILER})
  endif()
  if(CMAKE_GENERATOR)
    list(APPEND _host_args -G "${CMAKE_GENERATOR}")
  endif()
  if(CMAKE_GENERATOR STREQUAL "Ninja" AND CMAKE_MAKE_PROGRAM)
    list(APPEND _host_args -DCMAKE_MAKE_PROGRAM=${CMAKE_MAKE_PROGRAM})
  endif()
  message(STATUS "SM64: configuring host asset pipeline in ${SM64_HOST_ASSET_BINARY_DIR}")
  execute_process(COMMAND "${CMAKE_COMMAND}" ${_host_args} RESULT_VARIABLE _cfg_result)
  if(NOT _cfg_result EQUAL 0)
    message(FATAL_ERROR
      "SM64: host asset pipeline failed to configure (${_cfg_result}). "
      "Point SM64_HOST_C_COMPILER at a host compiler if the default one cannot build the tools.")
  endif()
  message(STATUS "SM64: building host assets (textures / sound / text) -> ${SM64_GEN_DIR}")
  execute_process(
    COMMAND "${CMAKE_COMMAND}" --build "${SM64_HOST_ASSET_BINARY_DIR}" --target sm64_assets
    RESULT_VARIABLE _build_result)
  if(NOT _build_result EQUAL 0)
    message(FATAL_ERROR "SM64: host asset pipeline failed to build (${_build_result}).")
  endif()
endfunction()

if(SM64_HOST_ASSET_BUILD)

# Run outside MSYS2 (RenderDoc, plain Explorer), where an unrelated
# libwinpthread-1.dll on PATH otherwise wins and fails to resolve nanosleep64.
# Windows prefers the exe's own directory, so copying the runtime beside it is
# enough -- and unlike -static it avoids MinGW's static winpthread, whose
# nanosleep disagrees with the headers about struct timespec.
if(MINGW)
  get_filename_component(_sm64_cxx_dir "${CMAKE_CXX_COMPILER}" DIRECTORY)
  foreach(_dll libgcc_s_seh-1.dll libstdc++-6.dll libwinpthread-1.dll libatomic-1.dll)
    if(EXISTS "${_sm64_cxx_dir}/${_dll}")
      add_custom_command(TARGET sm64_foundation POST_BUILD
        COMMAND "${CMAKE_COMMAND}" -E copy_if_different
                "${_sm64_cxx_dir}/${_dll}" "$<TARGET_FILE_DIR:sm64_foundation>"
        COMMENT "Staging ${_dll} next to sm64_foundation")
    endif()
  endforeach()
endif()
